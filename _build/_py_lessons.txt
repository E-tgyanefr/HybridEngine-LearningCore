# 第 03 课：列表与循环 —— 一次处理一堆东西


# ─────────────────────────────────────────────────────
#  为什么第 3 条要「返回一个新列表」
# ─────────────────────────────────────────────────────
#
# 新手最容易写成这样：
#
#     for x in 分数表:
#         if x < 60:
#             分数表.remove(x)      # ← 错！边遍历边删
#
# 这会**跳过**元素：删掉一个之后，后面的元素整体前移，
# 而 for 的下标已经走过去了 —— 于是紧挨着被删元素的那一项永远检查不到。
# 这个 bug 很隐蔽：数据少的时候看不出来，数据一多就漏。
#
# ⇒ 正确做法是**造一个新列表**（下面这个套路到处都是）：
#
#     出 = []
#     for x in 分数表:
#         if 条件:
#             出.append(x)
#     return 出
#
# 它还有个好处：**不修改传进来的东西**。
# 调用者的列表还是原样 —— 这叫「没有副作用」，是函数该有的样子。
#
# ─────────────────────────────────────────────────────
#  列表的四个常用操作
# ─────────────────────────────────────────────────────
#
#     成绩 = [90, 75, 60, 45, 88]
#     成绩[0]          第一个（下标从 0 开始，不是 1）
#     成绩[-1]         最后一个（负数从右边数）
#     len(成绩)        有几个
#     sum(成绩)        加起来（Python 自带的）
#     max(成绩)        最大的
#
#   ⚠ len / sum / max 是**内置函数**，不用自己写循环。
#     但这一课的 平均分 / 及格的人 / 最高分 请你**用循环写一遍** ——
#     先会自己写，再知道有现成的，顺序不能反。
#
# ─────────────────────────────────────────────────────
#
# ─────────────────────────────────────────────────────
#  轮到你了 —— 在下面的答题文件里写这几件事

"""第 03 课参考答案 —— 列表与循环。"""

成绩 = [90, 75, 60, 45, 88]


def 平均分(分数表):
    """平均分。用循环算一遍（虽然 Python 有 sum/len）。"""
    if not 分数表:              # ⚠ 空列表要特判，否则下面除以 0
        return 0.0
    和 = 0
    for x in 分数表:
        和 += x
    return 和 / len(分数表)      # ⚠ 一个斜杠 ⇒ 有小数；两个斜杠会丢小数


def 及格的人(分数表):
    """挑出及格的，**返回新列表**（不改传进来的那个）。"""
    出 = []
    for x in 分数表:
        if x >= 60:             # ⚠ 60 分是及格 ⇒ >= 不是 >
            出.append(x)
    return 出


def 最高分(分数表):
    """最大分数。"""
    # ⚠ 起点用**第一个元素**，不能写 0 —— 否则全负数时会返回 0
    最大 = 分数表[0]
    for x in 分数表:
        if x > 最大:
            最大 = x
    return 最大

# ─────────────────────────────────────────────────────
#  列表 vs 字典：什么时候用哪个
# ─────────────────────────────────────────────────────
#
#   列表：一排格子，按**位置**取 —— 成绩[0]、成绩[1]
#   字典：一堆抽屉，按**名字**取 —— 库存['苹果']
#
#   ⚠ 判断标准很简单：
#     你需要问「第几个」 ⇒ 列表
#     你需要问「叫什么」 ⇒ 字典
#
#   苹果 3 个、香蕉 5 个 —— 你不会想说「第 0 样是苹果」，
#   你只想说「苹果还有几个」。那就是字典。
#
#     库存 = {'苹果': 3, '香蕉': 5}
#     库存['苹果']            3
#     库存['橘子']            ← KeyError！没有这个键
#
# ─────────────────────────────────────────────────────
#  第 2 条为什么要「没有就返回 0」
# ─────────────────────────────────────────────────────
#
# 直接 库存[商品名] 在键不存在时会**抛 KeyError**，
# 而「查一个没有的商品」是**正常情况**，不该让程序崩。
#
# ⇒ 两种写法，都行：
#
#     库存.get(商品名, 0)              ← 最简（get 的第二个参数是「没有时给什么」）
#
#     if 商品名 in 库存:               ← 先问在不在
#         return 库存[商品名]
#     return 0
#
# 这一课请你**两种都写一遍**（第 2 条用 get，另外用 in 检查一次）。
#
# ─────────────────────────────────────────────────────
#  遍历字典
# ─────────────────────────────────────────────────────
#
#     for 名 in 库存:                  遍历**键**
#         print(名)
#
#     for 名, 数 in 库存.items():      同时拿到键和值（最常用）
#         print(名, 数)
#
#   ⚠ .items() 那一对括号不能省 —— 少了就变成遍历方法对象，会报错。
#
# ─────────────────────────────────────────────────────
#
# ─────────────────────────────────────────────────────

def CHECK_库存字典(答案):
    if not hasattr(答案, "库存"):
        return False, ("还没有定义「库存」这个字典。\n"
                       "      写法：库存 = {'苹果': 3, '香蕉': 5}\n"

    v = 答案.库存

        return False, f"「库存」里只有 {len(v)} 样商品，至少要 3 样。"
    if not all(isinstance(k, str) for k in v):

    if not all(isinstance(x, int) and not isinstance(x, bool) for x in v.values()):

    if len(set(v.values())) == 1:
        return False, ("所有商品数量都一样 —— 那样测不出「最多的是谁」。\n"
                       "      请让数量有差别。")
    return True, f"{len(v)} 样商品：{v}"

    for 名, 数 in 库存表.items():

# 第 05 课：函数 —— 把一件事打包，还能给默认值


# ─────────────────────────────────────────────────────
#  这一课三个语法点，都是「让函数好用」的
# ─────────────────────────────────────────────────────
#
#  ① **默认参数** —— 调用时可以不传
#
#         def 问候(名字, 称呼="同学"):
#             ...
#
#         问候("小明")            用默认的"同学"
#         问候("小明", "老师")     传了就用传的
#
#     ⚠ 有默认值的参数必须写在**后面**。
#        def f(a="x", b) 是语法错误。
#
#  ② **返回多个值** —— 其实返回的是一个**元组**
#
#         def 统计(数字们):
#             return len(数字们), sum(数字们)
#
#         个数, 总和 = 统计([1,2,3])      ← 这样「拆开」接住（叫解包）
#
#     ⚠ 不是真的返回了两个东西，而是返回了一个元组 (3, 6)，
#       左边用两个变量接的时候自动拆开。这一点在用到「整个返回值」时会露出来：
#         x = 统计([1,2,3])   ⇒ x 是 (3, 6)，不是 3
#
#  ③ **任意多个参数** —— 参数名前面加一个星号
#
#         def 加全部(*数字):
#             ...
#
#         加全部(1, 2, 3)      数字 是 (1, 2, 3)
#         加全部()             数字 是 ()   ← 空元组，要返回 0
#
#     ⚠ *数字 收进来的是**元组**，不是列表。
#       所以第 2 条返回元组、第 3 条接收元组 —— 这一课在练同一件事。
#
# ─────────────────────────────────────────────────────
#  为什么值得单独一课
# ─────────────────────────────────────────────────────
#
# 默认参数和 *args 不是「花哨语法」，它们决定了一个函数**好不好用**：
#
#     print(...)          能接任意多个 —— 所以你能一次打一堆东西
#     sorted(列表)        默认从小到大，但能 reverse=True 改 —— 所以常用
#
# 你以后写工具函数时，「要不要给个默认值」「要不要支持任意多个」
# 就是判断一个函数成熟不成熟的标准。
#
# ─────────────────────────────────────────────────────
#
# ─────────────────────────────────────────────────────
#  轮到你了 —— 在下面的答题文件里写这几件事
# ─────────────────────────────────────────────────────
#
#   1. 写函数 问候(名字, 称呼="同学")，返回 "你好，<名字><称呼>！"
#      例如 问候("小明")        → "你好，小明同学！"
#           问候("李老师", "老师") → "你好，李老师老师！"
#      ⚠ 「称呼」要**有默认值**，不传也能调用
#   2. 写函数 统计(数字们)，返回一个**元组** (个数, 总和)
#   3. 写函数 加全部(*数字)，接收**任意多个**参数，返回它们的和

# ─────────────────────────────────────────────────────
#  报错不是「你写错了」，是 Python 在告诉你哪里不对
# ─────────────────────────────────────────────────────
#
# 前五课你已经见过不少报错。这一课要学的是**另一面**：
# 有些「不对」是**正常的**，程序不该崩。
#
#     用户输入了 "abc"，你要转成整数 —— 转不了很正常，不该崩
#     用户想除以 0 —— 不该崩
#     想取第 100 个元素，而列表只有 3 个 —— 不该崩
#
# ⇒ 用 try / except 接住：
#
#     try:
#         可能出错的那句
#     except 具体的错误类型:
#         出错了怎么办
#
# ─────────────────────────────────────────────────────
#  ⚠⚠ 两条最重要的规矩
# ─────────────────────────────────────────────────────
#
#  ① **except 要写具体的错误类型**，不要空着。
#
#         except:               ← ❌ 什么都接
#         except Exception:     ← ⚠ 也很粗
#         except ValueError:    ← ✅ 明确
#
#     空的 except: 会把**你自己写错的代码**也一起吞掉 ——
#     那时程序不报错，但结果是错的，你会查很久。
#
#  ② **接住之后要给出一个合理的结果**，不是打个字就算了。
#
#         except ValueError:
#             return None       ← ✅ 调用者知道「没转成」
#
# 这一课三个函数要求的都是 return None ——
# 那是「我没做成」的通用说法，比抛异常温柔，也比返回 0 诚实
# （因为 0 是个合法结果，分不清是「转成了 0」还是「没转成」）。
#
# ─────────────────────────────────────────────────────
#  三种错误分别是什么
# ─────────────────────────────────────────────────────
#
#     int("abc")        ValueError          值不对（类型对，内容不对）
#     int(None)         TypeError           类型不对
#     10 / 0            ZeroDivisionError
#     [1,2][9]          IndexError          下标越界
#     {"a":1}["b"]      KeyError            字典里没这个键（第 04 课见过）
#
#   ⚠ IndexError 和 KeyError 是**不同**的类，别抓错。
#     拿不准时可以用 (IndexError, KeyError) 一次抓两个。
#
# ─────────────────────────────────────────────────────
#
# ─────────────────────────────────────────────────────
#  轮到你了 —— 在下面的答题文件里写这几件事
# ─────────────────────────────────────────────────────
#
#   1. 写函数 转成整数(文字)，把文字转成整数返回；
#      **转不了就返回 None**（不要让它抛异常）
#   2. 写函数 安全相除(a, b)，返回 a / b；
#      **b 是 0 就返回 None**
#   3. 写函数 取第几个(列表, 第几个)，返回那个元素；
#      **下标越界就返回 None**
#

# 第 07 课：类 —— 把数据和动作放在一起


# ─────────────────────────────────────────────────────
#  为什么需要「类」
# ─────────────────────────────────────────────────────
#
# 没有类的时候你会这么写：
#
#     余额 = 0
#     def 存(n):
#         global 余额        # ← 要声明 global，丑
#         余额 += n
#     def 取(n):
#         global 余额
#         余额 -= n          # ← 忘了检查够不够 ⇒ 余额变负数
#
# 两个问题：
#   ① `余额` 是个**谁都能改**的全局变量，别处一行代码就能弄坏它
#   ② 「余额」和「改余额的规则」**离得很远** —— 你想检查规则得翻遍整个文件
#
# 有类之后，它们被放进同一个地方：
#
#     class 存钱罐:
#         def __init__(self, 初值=0):
#             self.余额 = 初值
#         def 取(self, n):
#             if n > self.余额:        # ← 规则就在这里，跑不掉
#                 return False
#             self.余额 -= n
#             return True
#
# ⇒ 这叫**封装**。不要把它理解成「藏起来」，
#   它的真实意思是：**一个东西的规则只有一个出口**。
#   想把余额改成负数？做不到 —— 唯一的入口是 `取`，而 `取` 会检查。
#
# ─────────────────────────────────────────────────────
#  三个必须记住的东西
# ─────────────────────────────────────────────────────
#
#  ① `__init__` —— 造对象时自动跑的方法（前后各两个下划线）
#
#         def __init__(self, 初值=0):
#             self.余额 = 初值
#
#     ⚠ 不是 __int__、不是 _init_ —— 拼错的话它只是个普通方法，
#       造对象时**不会被调用**，然后你会奇怪「为什么属性不存在」。
#
#  ② `self` —— 「我自己」
#
#         每个方法第一个参数都是 self（名字可以改，但没人改）
#         写 self.余额 才是这个对象的余额；
#         写 余额 是一个**局部变量**，跟对象没关系 —— 新手最常混这点。
#
#  ③ 方法调用时**不用传 self**
#
#         定义：def 加(self, n):
#         调用：罐.加(5)          ← 不写 self，Python 自己塞进去
#
# ─────────────────────────────────────────────────────
#  第 2 条最容易错的地方
# ─────────────────────────────────────────────────────
#
# `取(n)` 钱不够时要「返回 False **且余额不变**」。
# 常见错法是**先扣再判断**：
#
#         self.余额 -= n            # ← 已经扣了！
#         if self.余额 < 0:
#             return False
#
# 这时余额已经被改坏了。正确顺序永远是：**先检查，通过才动**。
#
# ─────────────────────────────────────────────────────
#
# ─────────────────────────────────────────────────────
#  轮到你了 —— 在下面的答题文件里写这几件事
# ─────────────────────────────────────────────────────
#
#   1. 写一个类 计数器：
#        · 造出来时有 `值`（默认 0，可以传初值）
#        · 方法 `加(n)` 让值增加 n
#        · 方法 `重置()` 让值回到 0
#   2. 写一个类 存钱罐：
#        · 造出来时有 `余额`（默认 0）
#        · 方法 `存(n)` 增加余额
#        · 方法 `取(n)`：钱够就扣掉并返回 True；**不够就返回 False 且余额不变**

def CHECK_计数器类(答案):

        return False, f"「计数器」应当是类，现在是 {type(类).__name__}。"

        a = 类()
    except Exception as e:                                      # noqa: BLE001

                       "      ⚠ 一定要写 self.值 —— 只写 值 是个局部变量，对象上没有。")
    if a.值 != 0:
        return False, f"不传初值时 .值 应当是 0，实际是 {a.值!r}。"
    try:
        b = 类(100)
    except Exception as e:                                      # noqa: BLE001
        return False, (f"计数器(100) 造不出来：{type(e).__name__}: {e}\n"
                       f"      ⚠ 构造函数要有 初值=0 这个**带默认值**的参数。")
    if b.值 != 100:

    if not hasattr(a, "加") or not callable(a.加):
        return False, "没有 加(n) 这个方法。"
    a.加(3)
    a.加(4)
    if a.值 != 7:
        return False, f"加(3) 再 加(4) 之后应当是 7，实际是 {a.值!r}。"
    if not hasattr(a, "重置") or not callable(a.重置):
        return False, "没有 重置() 这个方法。"
    a.重置()

    c = 类()

                       f"        要写在 __init__ 里用 self.值 = ... 赋值。")
    return True, "默认初值 / 指定初值 / 加 / 重置 / 对象互不影响 都对"

"""第 08 课参考答案 —— 综合练习。"""


class 学生:
    def __init__(self, 姓名, 成绩表=None):
        self.姓名 = 姓名
        # ⚠⚠ 千万别写 成绩表=[]！
        #   默认的 [] 只在**定义时创建一次** ⇒ 所有不传成绩表的学生
        #   会共用同一个列表（给小明加分，小红也跟着有分）。
        #   正确写法：默认 None，进来再造列表。
        self.成绩表 = [] if 成绩表 is None else 成绩表

    def 加成绩(self, 分数):
        self.成绩表.append(分数)

    def 平均分(self):
        if not self.成绩表:         # ⚠ 空列表要特判，否则除以 0
            return 0
        return sum(self.成绩表) / len(self.成绩表)

    def 最高分(self):
        if not self.成绩表:
            return 0
        最大 = self.成绩表[0]        # ⚠ 起点用第一个元素，不能写 0（负数会错）
        for x in self.成绩表:
            if x > 最大:
                最大 = x
        return 最大

    def 评语(self):
        # ⚠ 顺序：先判「没有成绩」，再判分数档
        if not self.成绩表:
            return "还没有成绩"
        平均 = self.平均分()
        if 平均 >= 90:
            return "优秀"
        if 平均 >= 60:
            return "及格"
        return "要加油"


def 全班报告(学生表):
    """每个学生一行：姓名 平均分=xx.x 评语"""
    行们 = []
    for s in 学生表:
        行们.append(f"{s.姓名} 平均分={s.平均分():.1f} {s.评语()}")
    # ⚠ 用 "\n".join —— 空列表时正好返回空文字（边界自然对）
    return "\n".join(行们)
