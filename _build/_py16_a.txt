# 第 01 课：变量 —— 给一个值起个名字

# ─────────────────────────────────────────────────────
#  先介绍：变量是什么
# ─────────────────────────────────────────────────────
#
# 变量就是「**给一个值起个名字**」。
#
#     名字 = "小明"
#     ↑     ↑
#     名字   值
#
# 那个等号不是数学里的「相等」，是「**把右边的值装进左边的名字里**」。
# 读作「把 "小明" 赋给 名字」。
#
# ⚠ 起名字的意义在于**后面能重复使用**：
#
#     名字 = "小明"
#     print(名字)            ← 用到它，不用重打一遍
#     print(名字 + "你好")    ← 还能拼起来
#
# 如果不起名字，你就得每次重打一遍 "小明" ——
# 改的时候要改好几个地方，漏一处就错了。
#
# ─────────────────────────────────────────────────────
#  变量名怎么起
# ─────────────────────────────────────────────────────
#
#     年龄 = 18           ✅ 中文也行（Python 3 支持）
#     age = 18            ✅ 英文最常见
#     2号 = 18            ❌ 不能以数字开头
#     我的 名字 = 18       ❌ 不能有空格
#
#   ⚠ 大小写敏感：`名字` 和 `名字 ` 不是一回事，`Name` 和 `name` 也是两个变量。
#
# ─────────────────────────────────────────────────────
#  ⚠ 最常见的错：字符串忘了引号
# ─────────────────────────────────────────────────────
#
#     名字 = 小明          ❌ NameError: name '小明' is not defined
#     名字 = "小明"        ✅
#
#   Python 看到 `小明` 不带引号，会以为你在说「一个叫小明的变量」，
#   然后去找 —— 找不到就报 `NameError: name '小明' is not defined`。
#
#   ⇒ **错误信息里说 "name ... is not defined" 时，先看引号。**
#     这是新手第一个必踩的坑，也是这一课唯一要记住的坑。
#
# ─────────────────────────────────────────────────────
#  这一课只讲这一件事
# ─────────────────────────────────────────────────────
#   其他东西（字符串、数字、怎么打印）在下面几课，这一课不碰。
#
# ─────────────────────────────────────────────────────
#  轮到你了 —— 在答题文件里写这几件事
# ─────────────────────────────────────────────────────
#   1. 定义一个变量 我的名字，值是你自己的名字（**要加引号**）
#   2. 定义一个变量 我的年龄，值是你的年龄（整数，**不要引号**）
#
#   ⚠ 就这两条。不用打印、不用函数 —— 那是后面几课的事。
#
# ─────────────────────────────────────────────────────
#  怎么自己测
# ─────────────────────────────────────────────────────
#     python -m hybridengine_light 编辑 01     写代码
#     python -m hybridengine_light 检查 01     检查答案
#     python -m hybridengine_light 答案 01     参考答案（先自己写一遍）


def CHECK_我的名字(答案):
    if not hasattr(答案, "我的名字"):
        return False, ("还没有定义「我的名字」。\n"
                       "      写法：我的名字 = '小明'        ← 名字要加引号\n"
                       "      ⚠ 不加引号会报 NameError: name '小明' is not defined")
    v = 答案.我的名字
    if not isinstance(v, str):
        return False, (f"「我的名字」应当是**文字**（用引号括起来），现在是 {type(v).__name__}。\n"
                       f"      写法：我的名字 = '小明'\n"
                       f"      ⚠ 引号不能少 —— 少了 Python 会去找一个叫这个名字的变量。")
    if not v.strip():
        return False, "「我的名字」是空的 —— 写个真名字进去。"
    if len(v) > 40:
        return False, f"「我的名字」有 {len(v)} 个字符，太长了 —— 写个名字就行。"
    return True, f"我的名字 = {v!r}"


def CHECK_我的年龄(答案):
    if not hasattr(答案, "我的年龄"):
        return False, ("还没有定义「我的年龄」。\n"
                       "      写法：我的年龄 = 18          ← 整数**不要**引号\n"
                       "      ⚠ 写成 '18' 会是文字，不能拿来做算术（第 03 课会讲）。")
    v = 答案.我的年龄
    if isinstance(v, bool) or not isinstance(v, int):
        if isinstance(v, str):
            return False, (f"「我的年龄」现在是文字 {v!r} —— **把引号去掉**。\n"
                           f"      写法：我的年龄 = 18\n"
                           f"      ⚠ '18'（带引号）是文字，18（不带）才是数字。")
        return False, (f"「我的年龄」应当是整数，现在是 {type(v).__name__}。\n"
                       f"      写法：我的年龄 = 18")
    if not (0 < v < 200):
        return False, f"「我的年龄」是 {v} —— 这不像一个年龄。"
    return True, f"我的年龄 = {v}"

# 第 02 课：print 与字符串 —— 让程序开口说话

# ─────────────────────────────────────────────────────
#  先介绍：print 是什么
# ─────────────────────────────────────────────────────
#
# 前面写的程序，跑完什么都不会发生 —— 因为**你没让它说话**。
#
#     print("你好")
#     ↑
#     把括号里的东西**打印到屏幕上**
#
# 这是你和程序之间唯一的「输出」通道。没有 print，
# 你只能靠猜程序里发生了什么。
#
# ─────────────────────────────────────────────────────
#  字符串：一段文字
# ─────────────────────────────────────────────────────
#
# 用**引号**包起来的东西就是字符串：
#
#     "你好"        双引号
#     '你好'        单引号 —— 和双引号**完全一样**
#     "it's ok"     里面有单引号，就用双引号包
#     'he said "hi"'  里面有双引号，就用单引号包
#
#   ⚠ 两种引号没有区别，选哪种只看「里面有没有另一种引号」。
#
# ─────────────────────────────────────────────────────
#  ⚠ 三个最常见的坑
# ─────────────────────────────────────────────────────
#
#  ① **括号忘了**
#         print "你好"          ❌ Python 3 里 print 是函数，括号必须有
#         print("你好")          ✅
#
#  ② **中文标点**
#         print（"你好"）        ❌ 全角括号！报 SyntaxError: invalid character '（'
#         print("你好")          ✅ 括号和引号都要**半角**
#
#     这个坑特别隐蔽：全角括号和半角括号看起来几乎一样。
#     ⇒ 报 `invalid character` 时，先检查是不是中文输入法打的标点。
#
#  ③ **一次打印多个东西**
#         print("你好", "小明")     ⇒  你好 小明
#         print("你好" + "小明")    ⇒  你好小明
#     `,` 会加一个空格，`+` 直接拼 —— 不一样。
#
# ─────────────────────────────────────────────────────
#  这一课只讲这一件事
# ─────────────────────────────────────────────────────
#   怎么把变量放进句子里（f-string）是第 04 课，这一课不碰。
#
# ─────────────────────────────────────────────────────
#  轮到你了 —— 在答题文件里写这几件事
# ─────────────────────────────────────────────────────
#   1. 写一行 print，打印：第 02 课开始
#   2. 写一行 print，用 `+` 把两段文字拼起来打印：你好 + 世界
#
#   ⚠ 就这些。不用变量、不用函数。
#
# ─────────────────────────────────────────────────────
#  怎么自己测
# ─────────────────────────────────────────────────────
#     python -m hybridengine_light 检查 02


def _行(答案):
    """把答题文件的 print 输出抓成行。"""
    import io
    import contextlib
    import runpy
    buf = io.StringIO()
    try:
        with contextlib.redirect_stdout(buf):
            runpy.run_path(str(答案.__file__), run_name="__答题__")
    except Exception as e:                                      # noqa: BLE001
        return None, f"运行你的代码时出错了：{type(e).__name__}: {e}"
    return [l for l in buf.getvalue().split("\n") if l.strip()], None


def CHECK_打印了第一行(答案):
    if not hasattr(答案, "__file__"):
        return False, "找不到你的答题文件。"
    行, 错 = _行(答案)
    if 错:
        return False, 错
    if not 行:
        return False, ("你的文件一行都没打印。\n"
                       "      写法：print('第 02 课开始')\n"
                       "      ⚠ print 后面那对**圆括号**不能少。")
    if "第 02 课开始" not in 行[0]:
        return False, (f"第一行应当是「第 02 课开始」，实际是 {行[0]!r}。\n"
                       f"      ⚠ 检查：括号和引号是不是**半角**的？\n"
                       f"        全角（）会报 invalid character，全角引号也一样。")
    return True, f"第一行 = {行[0]}"


def CHECK_拼接打印了(答案):
    if not hasattr(答案, "__file__"):
        return False, "找不到你的答题文件。"
    行, 错 = _行(答案)
    if 错:
        return False, 错
    if len(行) < 2:
        return False, (f"应该有 2 行输出，实际 {len(行)} 行 —— 是不是忘了第二行 print？\n"
                       f"      写法：print('你好' + '世界')")
    if 行[1].strip() != "你好世界":
        return False, (f"第二行应当是「你好世界」（中间**没有空格**），实际是 {行[1]!r}。\n"
                       f"      ⚠ 用 `+` 拼接是紧挨着的；用 `,` 会多一个空格。\n"
                       f"        print('你好' + '世界')   ⇒  你好世界\n"
                       f"        print('你好', '世界')   ⇒  你好 世界")
    return True, f"第二行 = {行[1]}"

# 第 03 课：数字与算术 —— 让程序算起来

# ─────────────────────────────────────────────────────
#  先介绍：两种数字
# ─────────────────────────────────────────────────────
#
#     int     整数        18      0      -5      1000000
#     float   小数        3.14    0.5    -2.75    1.0
#
# 写的时候**带小数点就是 float，不带就是 int**：
#
#     a = 18        int
#     b = 18.0      float
#
# ─────────────────────────────────────────────────────
#  五个运算符
# ─────────────────────────────────────────────────────
#
#     +    加       7 + 2  ⇒  9
#     -    减       7 - 2  ⇒  5
#     *    乘       7 * 2  ⇒  14
#     /    除       7 / 2  ⇒  3.5      ← ⚠ 一定有小数
#     //   整除     7 // 2 ⇒  3        ← ⚠ 丢掉小数
#     %    取余     7 % 2  ⇒  1        ← 剩下多少
#     **   乘方     2 ** 10 ⇒ 1024
#
# ─────────────────────────────────────────────────────
#  ⚠⚠ 这一课唯一要记住的坑：`/` 和 `//` 不一样
# ─────────────────────────────────────────────────────
#
#     10 / 4    ⇒  2.5      ← 一个斜杠：老老实实除
#     10 // 4   ⇒  2        ← 两个斜杠：只留整数部分（**不做四舍五入**）
#
#   为什么容易错：`//` 看起来很像是"注释"或者"手滑多打了一个"，
#   于是你以为写的是除法 —— 结果小数没了，而程序**不报错**。
#
#   ⇒ 算钱、算平均分、算比例时，**必须用 `/`**。
#     一个丢掉小数的结果不会报错，只会算错 —— 那是最难查的一类 bug。
#
# ─────────────────────────────────────────────────────
#  ⚠ 文字和数字不能相加
# ─────────────────────────────────────────────────────
#
#     "18" + 1      ❌ TypeError: can only concatenate str (not "int") to str
#     18 + 1        ✅ 19
#     int("18") + 1 ✅ 20        ← int() 把文字转成整数
#
#   ⇒ 报 `TypeError` 且提到 `str` 和 `int` 时，就是这个问题。
#     （输入进来的东西**永远是文字**，要算就得先 int() —— 第 14 课会讲怎么安全地转。）
#
# ─────────────────────────────────────────────────────
#  这一课只讲这一件事
# ─────────────────────────────────────────────────────
#   怎么把算出来的结果显示出来，用 print —— 那是第 02 课。
#
# ─────────────────────────────────────────────────────
#  轮到你了 —— 在答题文件里写这几件事
# ─────────────────────────────────────────────────────
#   1. 定义变量 苹果数 = 7、梨数 = 2
#   2. 定义变量 总个数，值是 苹果数 + 梨数
#   3. 定义变量 平均，值是 总个数 / 2        ← ⚠ 一个斜杠
#
# ─────────────────────────────────────────────────────
#  怎么自己测
# ─────────────────────────────────────────────────────
#     python -m hybridengine_light 检查 03


def CHECK_两个数量(答案):
    for 名, 值 in (("苹果数", 7), ("梨数", 2)):
        if not hasattr(答案, 名):
            return False, f"还没有定义「{名}」。写法：{名} = {值}"
        v = getattr(答案, 名)
        if isinstance(v, bool) or not isinstance(v, int):
            return False, (f"「{名}」应当是整数 {值}，现在是 {type(v).__name__}：{v!r}\n"
                           f"      ⚠ 加了引号就成文字了。")
        if v != 值:
            return False, f"「{名}」应当是 {值}，现在是 {v}。"
    return True, "苹果数 = 7、梨数 = 2"


def CHECK_总个数(答案):
    if not hasattr(答案, "总个数"):
        return False, "还没有定义「总个数」。写法：总个数 = 苹果数 + 梨数"
    v = 答案.总个数
    if v != 9:
        return False, (f"「总个数」应当是 9（7 + 2），现在是 {v!r}。\n"
                       f"      ⚠ 要写成 `苹果数 + 梨数`，用两个**变量**相加，不是写死 9。")
    return True, "总个数 = 9"


def CHECK_平均(答案):
    if not hasattr(答案, "平均"):
        return False, "还没有定义「平均」。写法：平均 = 总个数 / 2"
    v = 答案.平均
    if v == 4:
        return False, ("「平均」算出来是 4 —— **小数被丢掉了**。\n"
                       "      ⚠⚠ 你用的是两个斜杠 `//`（整除），要用**一个**斜杠 `/`。\n"
                       "         9 // 2  ⇒  4      丢掉小数\n"
                       "         9 /  2  ⇒  4.5    正确\n"
                       "      这个错误**不会报错**，只会算错 —— 所以特别要小心。")
    if abs(v - 4.5) > 1e-9:
        return False, f"「平均」应当是 4.5（9 / 2），现在是 {v!r}。"
    return True, f"平均 = {v}（一个斜杠，小数保住了）"

# 第 04 课：f-string —— 把变量放进句子里

# ─────────────────────────────────────────────────────
#  先介绍：为什么需要它
# ─────────────────────────────────────────────────────
#
# 你想打印「你好，小明！」，而名字存在变量里。用第 02 课的 `+` 拼：
#
#     print("你好，" + 名字 + "！")
#
# 能行，但**括号和加号满天飞**，句子一长就看不清了。
# 而且变量不是文字时还得手动转：
#
#     print("我今年 " + str(年龄) + " 岁")      ← 不加 str() 会 TypeError
#
# ─────────────────────────────────────────────────────
#  f-string：在引号前面加一个 f
# ─────────────────────────────────────────────────────
#
#     print(f"你好，{名字}！")
#            ↑            ↑
#            f           变量名用花括号包起来
#
# 就这么多。`{...}` 里面可以直接写变量，也可以直接写算式：
#
#     f"{苹果数 + 梨数}"        ⇒  9
#     f"{年龄}"                 ⇒  18      ← **不用 str()**，f-string 自己会转
#
# ─────────────────────────────────────────────────────
#  ⚠⚠ 三个必须记住的坑
# ─────────────────────────────────────────────────────
#
#  ① **那个 f 不能少**
#
#         print("你好，{名字}！")      ❌  ⇒  你好，{名字}！
#         print(f"你好，{名字}！")     ✅  ⇒  你好，小明！
#
#     少了 f，花括号里的东西会**原样打出来** —— 不报错，只是不对。
#
#  ② **花括号要用半角** `{` `}`，不能用全角 `｛` `｝`
#     而且里面**不要加引号**：`f"{'名字'}"` 是「打印"名字"这两个字」，
#     不是打印变量 —— 因为加了引号它就变成一段文字了。
#
#  ③ **变量名必须真的存在**
#     `f"{名子}"`（打错字）⇒  NameError: name '名子' is not defined
#     这时照着错误信息里的名字，检查你拼写的**每一个字**。
#
# ─────────────────────────────────────────────────────
#  还可以控制小数位数（顺便记住）
# ─────────────────────────────────────────────────────
#
#     f"{3.14159:.2f}"     ⇒  3.14        （保留 2 位小数）
#     f"{4.5:.1f}"         ⇒  4.5
#
#   冒号后面是"格式"，`.2f` 意思是"小数点后 2 位的浮点数"。
#
# ─────────────────────────────────────────────────────
#  这一课只讲这一件事
# ─────────────────────────────────────────────────────
#   print 本身是第 02 课；数字运算在第 03 课。这一课只碰 f-string。
#
# ─────────────────────────────────────────────────────
#  轮到你了 —— 在答题文件里写这几件事
# ─────────────────────────────────────────────────────
#   1. 定义变量 名字（文字）和 年龄（整数）
#   2. 写一个函数 自我介绍()，**用 f-string** 返回：
#         我叫<名字>，今年<年龄>岁。
#
# ─────────────────────────────────────────────────────
#  怎么自己测
# ─────────────────────────────────────────────────────
#     python -m hybridengine_light 检查 04


def CHECK_两个变量(答案):
    for 名, 类型, 说明 in (("名字", str, "文字（要加引号）"),
                           ("年龄", int, "整数（不要引号）")):
        if not hasattr(答案, 名):
            return False, f"还没有定义「{名}」（{说明}）。"
        v = getattr(答案, 名)
        if isinstance(v, bool) or not isinstance(v, 类型):
            return False, (f"「{名}」应当是{说明}，现在是 {type(v).__name__}：{v!r}")
    if not 答案.名字.strip():
        return False, "「名字」是空的。"
    return True, f"名字 = {答案.名字!r}，年龄 = {答案.年龄}"


def CHECK_自我介绍(答案):
    if not hasattr(答案, "自我介绍"):
        return False, "还没有定义「自我介绍」函数。写法：def 自我介绍(): ..."
    if not callable(答案.自我介绍):
        return False, "「自我介绍」不是函数。"
    try:
        out = 答案.自我介绍()
    except Exception as e:                                      # noqa: BLE001
        return False, (f"自我介绍() 出错了：{type(e).__name__}: {e}\n"
                       f"      ⚠ 如果是 NameError，照着错误里的名字检查拼写。")
    if not isinstance(out, str):
        return False, (f"「自我介绍」应当**返回文字**，现在返回的是 {type(out).__name__}。\n"
                       f"      写法：return f\"我叫{答案.名字}，今年{答案.年龄}岁。\"")
    期望 = f"我叫{答案.名字}，今年{答案.年龄}岁。"
    if out != 期望:
        # 少了 f 的典型症状

# 第 06 课：列表的增删改 —— 往格子里放东西、拿走东西

# ─────────────────────────────────────────────────────
#  先介绍：列表是"活的"
# ─────────────────────────────────────────────────────
#
# 第 05 课只讲了"取"。但列表真正的用处是**一边跑一边变**：
# 记分、排队、购物车 —— 都是往列表里加、从列表里减。
#
# ─────────────────────────────────────────────────────
#  四个动作
# ─────────────────────────────────────────────────────
#
#     购物 = ["苹果"]
#
#   ① 加一个到**末尾**
#        购物.append("香蕉")        ⇒  ["苹果", "香蕉"]
#
#   ② 插到**指定位置**
#        购物.insert(0, "牛奶")     ⇒  ["牛奶", "苹果", "香蕉"]
#                                     ↑ 插在下标 0 的位置，原来的往后挪
#
#   ③ 按**值**删掉一个
#        购物.remove("苹果")        ⇒  ["牛奶", "香蕉"]
#                                     ⚠ 值不存在会 KeyError 之外 —— 是 ValueError
#
#   ④ 按下标**弹出**（删掉并拿到它）
#        得 = 购物.pop()            ⇒  得 = "香蕉"，购物变成 ["牛奶"]
#        得 = 购物.pop(0)           ⇒  得 = "牛奶"，购物变成 []
#
#   ⚠ 还有一个直接改：
#        购物[0] = "果汁"            ⇒  把第 0 个换掉
#
# ─────────────────────────────────────────────────────
#  ⚠⚠ 这一课唯一要记住的坑：边遍历边删会漏元素
# ─────────────────────────────────────────────────────
#
#     for x in 分数表:
#         if x < 60:
#             分数表.remove(x)        # ← ❌ 错！
#
#   为什么错：删掉一个之后，**后面的元素整体往前挪一格**，
#   而 `for` 的下标已经走过去了 —— 于是紧挨着被删元素的那一项
#   **永远检查不到**。
#
#   看个例子（[50, 30, 90] 想删掉不及格的）：
#
#     第 1 轮：下标 0，值是 50 ⇒ 删掉。列表变成 [30, 90]
#     第 2 轮：下标 1，值是 90 ⇒ 30 被跳过了！**没删掉**
#
#   ⇒ 结果 [30, 90] —— 30 还在，而程序**不报错**。
#
#   正确做法有两种：
#
#     ① 造一个新列表（推荐，最不容易错）
#          出 = []
#          for x in 分数表:
#              if x >= 60:
#                  出.append(x)
#          return 出
#
#     ② 从后往前删（倒着遍历就不会被"前移"影响）
#          for i in range(len(分数表) - 1, -1, -1):
#              if 分数表[i] < 60:
#                  分数表.pop(i)
#
#   ⚠ 第 ① 种还有个好处：**不改调用者的列表**。
#     调用者手里的数据莫名其妙变少，是很讨厌的一件事。
#
# ─────────────────────────────────────────────────────
#  这一课只讲这一件事
# ─────────────────────────────────────────────────────
#   循环（for）是第 08 课。这一课先用**最简单的** for 来演示那个坑，
#   你只要记住"别边遍历边删"就行，不用现在就精通循环。
#
# ─────────────────────────────────────────────────────
#  轮到你了 —— 在答题文件里写这几件事
# ─────────────────────────────────────────────────────
#   1. 定义列表 购物 = ["苹果"]
#   2. 用 append 加一个 "香蕉"
#   3. 用 insert 在下标 0 插入 "牛奶"
#   4. 用 remove 删掉 "苹果"
#   5. 定义变量 弹出的，值是 购物.pop() 的结果
#

# 第 07 课：条件判断 —— 让程序做选择

# ─────────────────────────────────────────────────────
#  先介绍：程序要会"分岔"
# ─────────────────────────────────────────────────────
#
# 到现在为止，你的程序都是一条直线：从第一行跑到最后一行。
# 但真实的事情几乎都需要判断：
#
#     分数 >= 60 才及格
#     余额够才让取钱
#     名字为空就提示"请输入名字"
#
# 判断用 `if`：
#
#     if 条件:
#         条件成立时做的事          ← ⚠ 这一行要**缩进**
#
# ─────────────────────────────────────────────────────
#  缩进是语法，不是美观
# ─────────────────────────────────────────────────────
#
#     if 年龄 >= 18:
#     print("成年")            ❌ IndentationError: expected an indented block
#
#     if 年龄 >= 18:
#         print("成年")        ✅ 缩进 4 个空格
#
#   ⚠ Python 靠**缩进**判断"哪些行属于这个 if"。
#     其他语言用花括号 {}，Python 用空格 —— 这是 Python 最鲜明的特点。
#     统一用 **4 个空格**（别用 Tab，也别混用）。
#
# ─────────────────────────────────────────────────────
#  三种形态
# ─────────────────────────────────────────────────────
#
#   ① 只有 if
#         if 分数 >= 60:
#             print("及格")
#
#   ② if + else（两条路必走一条）
#         if 分数 >= 60:
#             print("及格")
#         else:
#             print("不及格")
#
#   ③ if + elif + else（多条路）
#         if 分数 >= 90:
#             等级 = "优秀"
#         elif 分数 >= 60:
#             等级 = "及格"
#         else:
#             等级 = "不及格"
#
# ─────────────────────────────────────────────────────
#  比较运算符
# ─────────────────────────────────────────────────────
#
#     ==   等于        ⚠ **两个等号**！一个等号是赋值
#     !=   不等于
#     >    大于        >=  大于或等于
#     <    小于        <=  小于或等于
#
#   ⚠⚠ 最常见的错：
#
#         if 分数 = 60:            ❌ SyntaxError
#         if 分数 == 60:           ✅
#
#     一个等号是"把 60 装进 分数"（赋值），两个才是"相等吗"。
#
# ─────────────────────────────────────────────────────
#  ⚠ 判断的顺序很重要
# ─────────────────────────────────────────────────────
#
#     if 分数 >= 60:
#         等级 = "及格"
#     elif 分数 >= 90:            ← ❌ 到不了这里！
#         等级 = "优秀"
#
#   95 分在第一行就已经满足 `>= 60` 了，程序不会再看后面的。
#   ⇒ **条件要从"最严格的"往"最宽松的"写**。
#
# ─────────────────────────────────────────────────────
#  这一课只讲这一件事
# ─────────────────────────────────────────────────────
#   循环在第 08 课。这一课只写"分岔"，不写循环。
#
# ─────────────────────────────────────────────────────
#  轮到你了 —— 在答题文件里写这几件事
# ─────────────────────────────────────────────────────
#   写一个函数 判等级(分数)：
#     · 90 分及以上  → 返回 "优秀"
#     · 60 分及以上  → 返回 "及格"
#     · 其他         → 返回 "不及格"
#
#   ⚠ 检查会试 95 / 90 / 60 / 59 四个分数，特别注意 **90 和 60** 这两个边界。
#
# ─────────────────────────────────────────────────────

# 第 08 课：循环 —— 让程序重复做事

# ─────────────────────────────────────────────────────
#  先介绍：为什么需要循环
# ─────────────────────────────────────────────────────
#
# 要打印 5 个分数，不要循环就得写 5 行 print。
# 一百个分数就写一百行 —— 而且加一个就要改一次代码。
#
# 循环让**一段代码重复跑**：
#
#     for x in 分数表:
#         print(x)
#         ↑ 这一行会被执行 len(分数表) 次
#
# ─────────────────────────────────────────────────────
#  for 的两半，各有名字
# ─────────────────────────────────────────────────────
#
#     for  x  in  分数表:
#          ↑        ↑
#        取出来的   要遍历的东西
#        那个东西
#
#   读作：「**对于 分数表 里的每一个元素，把它叫做 x，然后做下面的事**」。
#
#   ⚠ `x` 这个名字是你随便起的（`分数`、`item`、`每个` 都行），
#     但**要起得看得懂**：`for x in ...` 在复杂代码里是灾难。
#
# ─────────────────────────────────────────────────────
#  两种常见的循环
# ─────────────────────────────────────────────────────
#
#   ① 遍历一个列表（最常用）
#
#         for 分数 in 分数表:
#             print(分数)
#
#   ② 重复固定次数
#
#         for i in range(5):
#             print("第", i, "次")        ⇒ 0 1 2 3 4     ← ⚠ 从 0 开始！
#
#         range(1, 6)                     ⇒ 1 2 3 4 5   ← 含头不含尾
#
#   ⚠ `range(5)` 给的是 0,1,2,3,4 —— **5 个**数，但没有 5。
#     `range(1, 6)` 才是 1 到 5。这两个最容易记混。
#
# ─────────────────────────────────────────────────────
#  累加：循环最常用的套路
# ─────────────────────────────────────────────────────
#
#     和 = 0                      ← ⚠ 起点**必须**先定义
#     for x in [1, 2, 3]:
#         和 += x                 ← 和 = 和 + x 的简写
#     print(和)                   ⇒ 6
#
#   ⚠ 忘了 `和 = 0` 会报 `NameError`（第 01 课见过这个错）。
#
# ─────────────────────────────────────────────────────
#  这一课只讲这一件事
# ─────────────────────────────────────────────────────
#   这一课**只遍历**、不修改列表（增删是第 06 课）。
#   函数（def）是第 09 课，这一课先不写函数。
#
# ─────────────────────────────────────────────────────
#  轮到你了 —— 在答题文件里写这几件事
# ─────────────────────────────────────────────────────
#   1. 定义列表 数字 = [3, 1, 4, 1, 5]
#   2. 定义变量 和（先给 0），然后用 for 把 数字 里的每个数累加进去
#   3. 定义变量 个数，用循环数出有几个数（**不要用 len()**，练循环）
#
# ─────────────────────────────────────────────────────

# 第 09 课：函数 —— 把一段代码打包起来，起个名字

# ─────────────────────────────────────────────────────
#  先介绍：为什么要打包
# ─────────────────────────────────────────────────────
#
# 假设你要算三次平均分，就得把同样的几行抄三遍：
#
#     和 = 0
#     for x in 成绩1: 和 += x
#     平均1 = 和 / len(成绩1)
#
#     # ... 又抄一遍 ... 又抄一遍 ...
#
# 抄三遍的问题不是"打字数多"，而是**改的时候要改三处** ——
# 漏一处就出 bug。而且三个月后你看到这段代码，得重新读一遍才知道它在干嘛。
#
# 函数把这几行**打包**并**起个名字**：
#
#     def 算平均(分数表):
#         和 = 0
#         for x in 分数表:
#             和 += x
#         return 和 / len(分数表)
#
# 以后只要写 `算平均(成绩1)` —— 一眼就知道在做什么。
#
# ─────────────────────────────────────────────────────
#  def 的四个部分
# ─────────────────────────────────────────────────────
#
#     def  算平均  (  分数表  ):
#      ↑      ↑        ↑      ↑
#    关键字  函数名   参数    冒号不能少
#
#     函数体要**缩进**（和 if / for 一样，4 个空格）
#
# ─────────────────────────────────────────────────────
#  参数：把外面的东西递进去
# ─────────────────────────────────────────────────────
#
#     def 加一(数):
#         return 数 + 1
#
#     加一(5)        ⇒  6          这里的 5 会装进 `数`
#
#   ⚠ 参数名（`数`）和调用时传的东西（`5`）是**两回事**：
#     参数名只在函数**里面**有效，外面看不见。
#
#   ⚠ 参数个数要对上：
#         加一()         ❌ TypeError: 加一() missing 1 required positional argument
#         加一(5, 6)     ❌ TypeError: takes 1 positional argument but 2 were given
#     报这类 `TypeError` 时，数一数参数个数。
#
# ─────────────────────────────────────────────────────
#  ⚠⚠ 这一课唯一要记住的坑：光定义不会执行
# ─────────────────────────────────────────────────────
#
#     def 打招呼():
#         print("你好")
#
#     # 到这里，程序**什么都没打印** —— 你只是"教"了 Python 一个新动作，
#     # 还没让它做。
#
#     打招呼()        ← 这一行才是"让它做"
#
#   ⇒ 现象：定义了一堆函数，跑起来什么输出都没有。
#     原因永远是：**忘了调用**（没写那一对括号）。
#
# ─────────────────────────────────────────────────────
#  这一课只讲这一件事
# ─────────────────────────────────────────────────────
#   `return`（把结果交出来）是第 10 课。这一课的函数可以先只做动作、
#   或者在最后用 print 把结果打出来 —— 不用 return。
#
# ─────────────────────────────────────────────────────
#  轮到你了 —— 在答题文件里写这几件事
# ─────────────────────────────────────────────────────
#   1. 写函数 打招呼()：打印两行
#          你好
#          我在学 Python
#   2. **调用它一次**（检查会看你有没有真的调用）
#
#   ⚠ 就这两条。不用 return、不用参数 —— 那是第 10、11 课。
#
# ─────────────────────────────────────────────────────
#  怎么自己测
# ─────────────────────────────────────────────────────

def _跑文件(答案):
    import io
    import contextlib
    import runpy
    buf = io.StringIO()
    try:
        with contextlib.redirect_stdout(buf):
            runpy.run_path(str(答案.__file__), run_name="__答题__")
    except Exception as e:                                      # noqa: BLE001
        return None, f"运行你的代码时出错了：{type(e).__name__}: {e}"
    return [l for l in buf.getvalue().split("\n") if l.strip()], None

# 第 10 课：return —— 把结果交出来

# ─────────────────────────────────────────────────────
#  先介绍：print 和 return 完全不是一回事
# ─────────────────────────────────────────────────────
#
# 这是初学者最大的一个坎。看两个函数：
#
#     def 甲(x):
#         print(x * 2)          ← 只是**打出来给你看**
#
#     def 乙(x):
#         return x * 2          ← **把结果交出去**，给调用它的人
#
# 用起来差别巨大：
#
#     甲(5)            ⇒ 屏幕上是 10，但你**拿不到**这个 10
#     得 = 甲(5)        ⇒ 得 是 **None**！（函数没交东西出来）
#
#     乙(5)            ⇒ 屏幕上**什么都没有**
#     得 = 乙(5)        ⇒ 得 是 10 ✅
#
#   ⇒ 一句话：**print 是给人看的，return 是给程序用的。**
#
#   什么时候用哪个：
#     · 只想让你看到 → print
#     · 结果还要接着算（再相加、再比较、存起来）→ **return**
#
# ─────────────────────────────────────────────────────
#  ⚠⚠ 这一课唯一的坑：忘了 return，拿到 None
# ─────────────────────────────────────────────────────
#
#     def 算平均(分数表):
#         和 = sum(分数表) / len(分数表)
#                                   ← 没写 return！
#
#     算平均([60, 80])        ⇒  None
#     算平均([60, 80]) + 1    ⇒  TypeError: unsupported operand type(s) for +: 'NoneType' and 'int'
#
#   ⇒ 看到 **`NoneType`** 出现在报错里，八成就是"某个函数忘了 return"。
#     这是最值得记住的一条排错线索。
#
# ─────────────────────────────────────────────────────
#  return 还有一个作用：立刻结束函数
# ─────────────────────────────────────────────────────
#
#     def 判等级(分数):
#         if 分数 >= 90:
#             return "优秀"          ← 到这里就结束了
#         return "其他"              ← 上面返回过就不会走到这里
#
#   ⚠ `return` 之后的代码**不会被执行**。写多了也是白写（而且会让人看不懂）。
#
# ─────────────────────────────────────────────────────
#  一个函数可以返回多个值 —— 其实是一个"元组"
# ─────────────────────────────────────────────────────
#
#     def 统计(数字们):
#         return len(数字们), sum(数字们)
#              ↑ 用逗号隔开
#
#     个数, 总和 = 统计([1, 2, 3])      ⇒  个数=3，总和=6      （自动拆开）
#     整个 = 统计([1, 2, 3])            ⇒  整个=(3, 6)        （其实是一个东西）
#
#   ⚠ 不是真的返回两个，而是返回了一个**元组** `(3, 6)`。
#     左边用两个变量接的时候自动拆开 —— 这叫"解包"。
#
# ─────────────────────────────────────────────────────
#  这一课只讲这一件事
# ─────────────────────────────────────────────────────
#   带默认值的参数是第 11 课，接收任意多个是第 12 课。
#
# ─────────────────────────────────────────────────────
#  轮到你了 —— 在答题文件里写这几件事
# ─────────────────────────────────────────────────────
#   1. 写函数 平方(数)，**返回**它的平方
#   2. 写函数 统计(数字们)，**返回**一个元组：(个数, 总和)
#
#   ⚠ 两个都要用 return，不要用 print —— 检查会拿走返回值去比对。
#
# ─────────────────────────────────────────────────────

def CHECK_平方(答案):
    if not hasattr(答案, "平方"):
        return False, "还没有定义「平方」函数。写法：def 平方(数): return 数 * 数"
    if not callable(答案.平方):
        return False, "「平方」不是函数。"
    for n, 期望 in ((3, 9), (0, 0), (-4, 16), (1, 1)):
        try:
            得 = 答案.平方(n)
        except Exception as e:                                  # noqa: BLE001

# 第 11 课：默认参数 —— 调用时可以少传几个

# ─────────────────────────────────────────────────────
#  先介绍：让函数"不用每次都填满"
# ─────────────────────────────────────────────────────
#
#      def 问候(名字, 称呼):
#          return f"你好，{名字}{称呼}！"
#
#     问候("小明", "同学")      每次都得写两样。
#     可大部分时候称呼就是"同学" —— 每次都写很烦。
#
# 给参数一个**默认值**：
#
#     def 问候(名字, 称呼="同学"):
#                         ↑ 用等号给默认值
#         return f"你好，{名字}{称呼}！"
#
#     问候("小明")             ⇒ 你好，小明同学！        （用默认的）
#     问候("小明", "老师")      ⇒ 你好，小明老师！        （用传进来的）
#
#   ⇒ 不传就用默认值，传了就用你给的。
#
# ─────────────────────────────────────────────────────
#  ⚠⚠ 这一课唯一要记住的坑：有默认值的参数必须写在后面
# ─────────────────────────────────────────────────────
#
#     def 问候(称呼="同学", 名字):      ❌ SyntaxError
#                                      non-default argument follows default argument
#
#     def 问候(名字, 称呼="同学"):      ✅
#
#   为什么：如果默认值在前面，调用时 `问候("小明")` 到底是给名字还是给称呼？
#   Python 规定不了，所以**直接不允许这么写**。
#
#   ⇒ 报 `non-default argument follows default argument` 时，
#     把带默认值的参数**挪到后面**。
#
# ─────────────────────────────────────────────────────
#  ⚠⚠ 另一个更阴的坑：默认值别用列表
# ─────────────────────────────────────────────────────
#
#     def 记一笔(事情, 清单=[]):        ❌ 千万别这么写！
#         清单.append(事情)
#         return 清单
#
#   看起来没问题，但默认的 `[]` **只在定义函数时创建一次**，
#   所有调用会**共用同一个列表**：
#
#     记一笔("买菜")     ⇒ ['买菜']
#     记一笔("取快递")   ⇒ ['买菜', '取快递']     ← 上一次的还在！
#
#   ⇒ 正确写法是用 `None` 当默认值，进函数再造：
#
#     def 记一笔(事情, 清单=None):
#         if 清单 is None:
#             清单 = []
#         清单.append(事情)
#         return 清单
#
#   这个坑叫「**可变默认参数**」，是 Python 最有名的坑之一。

#

#  这一课只讲这一件事
# ─────────────────────────────────────────────────────

# 第 12 课：*参数 —— 想传几个就传几个

# ─────────────────────────────────────────────────────
#  先介绍：参数个数可以不定
# ─────────────────────────────────────────────────────
#
# 你一直在用 `print`，有没有想过它怎么能接任意多个东西？
#
#     print("a")
#     print("a", "b")
#     print("a", "b", "c", "d", ...)      都能用！
#
# 因为它用了 `*参数`：
#
#     def 加全部(*数字):
#         ...
#
# 那个**星号**的意思是：「把传进来的所有参数**收成一个元组**」。
#
#     加全部(1, 2, 3)      数字 是 (1, 2, 3)
#     加全部(5)            数字 是 (5,)
#     加全部()             数字 是 ()          ← 空元组，不是报错
#
# ─────────────────────────────────────────────────────
#  ⚠ 收进来的是**元组**，不是列表
# ─────────────────────────────────────────────────────
#
#     type(数字)      ⇒  <class 'tuple'>
#
#   能用 for 遍历（和列表一样），但**不能** append（元组不可改）。
#   ⇒ 要加东西就先转成列表：`表 = list(数字)`
#
# ─────────────────────────────────────────────────────
#  ⚠⚠ 这一课唯一要记住的坑：一个都不传时怎么办
# ─────────────────────────────────────────────────────
#
#     def 加全部(*数字):
#         和 = 0
#         for x in 数字:
#             和 += x
#         return 和
#
#   `加全部()` 时 `数字` 是 `()` —— 循环一次都不跑，返回 0。**这是对的。**
#
#   但如果你写成这样：
#
#     def 最大值(*数字):
#         return max(数字)         ← 加全部() 时会 ValueError: max() arg is an empty sequence
#
#   ⇒ 用 `*参数` 时**必须想一下"什么都不传"是什么意思**。
#     能返回 0 就返回 0；不能返回 0 的（比如最大值），要想清楚返回什么。
#
# ─────────────────────────────────────────────────────
#  顺带：星号还有一种用法（反过来的）
# ─────────────────────────────────────────────────────
#
#     def 画点(x, y):
#         ...
#
#     坐标 = (3, 4)
#     画点(*坐标)          ⇒  等于 画点(3, 4)      ← 把元组"拆开"传进去
#
#   同一个星号，在**定义**时是"收起来"，在**调用**时是"拆开"。
#
# ─────────────────────────────────────────────────────
#  这一课只讲这一件事
# ─────────────────────────────────────────────────────
#   这一课只讲 `*`。关键字参数（`**`）等你以后用到再学，现在不用碰。
#

# 第 13 课：字典 —— 按名字找东西

# ─────────────────────────────────────────────────────
#  先介绍：列表不够用的时候
# ─────────────────────────────────────────────────────
#
# 列表按**位置**取：`成绩[0]`、`成绩[1]`。
# 可有些东西你不想问"第几个"，只想问"叫什么"：
#
#     苹果还有几个？        香蕉呢？
#
# 列表里你就得记住苹果在第几位 —— 而且商品一多就记不住了。
#
# 字典按**名字**取：
#
#     库存 = {"苹果": 3, "香蕉": 5}
#             ↑      ↑
#             键     值
#
#     库存["苹果"]      ⇒  3
#
# ─────────────────────────────────────────────────────
#  什么时候用列表、什么时候用字典
# ─────────────────────────────────────────────────────
#
#     你要问「**第几个**」   ⇒  列表     成绩[0]、成绩[1]
#     你要问「**叫什么**」   ⇒  字典     库存["苹果"]
#
#   ⚠ 一句话记法：**一排格子用列表，一堆抽屉用字典。**
#
# ─────────────────────────────────────────────────────
#  三种常用操作
# ─────────────────────────────────────────────────────
#
#     库存 = {"苹果": 3}
#
#   ① 加 / 改（写法一样 —— 有就改，没有就加）
#        库存["香蕉"] = 5         ⇒  {"苹果": 3, "香蕉": 5}
#        库存["苹果"] = 10        ⇒  {"苹果": 10, "香蕉": 5}
#
#   ② 查
#        库存["苹果"]            ⇒  10
#        库存["橘子"]            ❌ KeyError: '橘子'
#
#   ③ 遍历
#        for 名 in 库存:                  遍历的是**键**
#        for 名, 数 in 库存.items():      同时拿键和值 ← 最常用
#
#   ⚠ `.items()` 那对括号不能省 —— 少了就变成遍历一个方法对象。

# 第 14 课：try / except —— 会出错，也要会处理出错

# ─────────────────────────────────────────────────────
#  先介绍：有些"不对"是正常的
# ─────────────────────────────────────────────────────
#
# 前面几课你见过的报错，都是"你写错了"。
# 但有些"不对"是**正常情况**，程序不该崩：
#
#     用户输入 "abc"，你要转成整数     —— 转不了很正常
#     用户想除以 0                    —— 不该崩
#     想取第 100 个元素，而列表只有 3 个 —— 不该崩
#
# 处理办法是 `try / except`：
#
#     try:
#         可能出错的那句
#     except 具体的错误类型:
#         出错了怎么办
#
#   ⇒ 读作：「**试着做这个；要是出了这种错，就改做那个。**」
#
# ─────────────────────────────────────────────────────
#  常见错误类型（要认得）
# ─────────────────────────────────────────────────────
#
#     ValueError         值不对          int("abc")
#     TypeError          类型不对        int(None)、"18" + 1
#     ZeroDivisionError  除以 0          10 / 0
#     IndexError         下标越界        [1,2][9]
#     KeyError           字典没这个键    {"a":1}["b"]
#     NameError          名字没定义      引号忘了 / 拼错了
#
#   ⚠ IndexError 和 KeyError 是**不同**的类，别抓错。
#
# ─────────────────────────────────────────────────────
#  ⚠⚠ 这一课唯一要记住的规矩：except 要写具体类型
# ─────────────────────────────────────────────────────
#
#     try:
#         ...
#     except:                  ❌ 什么都接 —— **千万别**
#     except Exception:        ⚠ 也很粗
#     except ValueError:       ✅ 明确
#
#   为什么空的 `except:` 危险：它会把**你自己写错的代码**也一起吞掉。
#
#     try:
#         成绩 = [90, 75]
#         print(成绩[5])            ← 你写错了（越界）
#     except:
#         print("出错了")            ← 被你接住了，程序不报错
#
#   于是你以为"越界这件事已经处理好了"，其实是**你写错了下标**。
#   程序不报错但结果不对 —— 那是最难查的一类问题。
#
#   ⇒ **只接你预料到的那种错误。**
#
# ─────────────────────────────────────────────────────
#  接住之后要给一个合理的结果
# ─────────────────────────────────────────────────────
#
#     except ValueError:
#         return None          ← "我没做成"的通用说法
#
#   ⚠ 为什么不用 0？因为 0 是个**合法结果**，
#     调用者分不清"转成了 0"还是"没转成"。
#     `None` 是"什么都没有"，不会有这个歧义。
#
# ─────────────────────────────────────────────────────
#  这一课只讲这一件事
# ─────────────────────────────────────────────────────
#   这一课只讲怎么"接住"错误。怎么**抛出**错误（raise）以后再说。
#
# ─────────────────────────────────────────────────────
#  轮到你了 —— 在答题文件里写这几件事
# ─────────────────────────────────────────────────────
#   1. 写函数 转成整数(文字)：能转就返回整数，**转不了返回 None**（不能崩）
#   2. 写函数 安全相除(a, b)：返回 a / b，**b 是 0 就返回 None**
#
# ─────────────────────────────────────────────────────
#  怎么自己测
# ─────────────────────────────────────────────────────
#     python -m hybridengine_light 检查 14

# 第 15 课：类与 __init__ —— 造一个属于你自己的东西

# ─────────────────────────────────────────────────────
#  先介绍：为什么需要"类"
# ─────────────────────────────────────────────────────
#
# 到现在你的数据和方法是**分开**的：
#
#     余额 = 0
#     def 存(n):
#         global 余额             ← 要声明 global，丑
#         余额 += n
#
# 问题是「余额」这个数据和「怎么改余额」这件事**离得很远**，
# 而且 `余额` 是个谁都能改的全局变量。
#
# 类把**数据**和**操作数据的方法**放进同一个地方：
#
#     class 存钱罐:
#         def __init__(self, 初值=0):
#             self.余额 = 初值
#
#         def 存(self, n):
#             self.余额 += n
#
#     罐 = 存钱罐(100)          ← 造一个
#     罐.存(50)                 ← 用它的方法
#     罐.余额                   ⇒ 150
#
# ─────────────────────────────────────────────────────
#  `__init__` 是什么
# ─────────────────────────────────────────────────────
#
# 它是"**造对象时自动跑一次**"的方法，用来把东西准备好：
#
#     class 计数器:
#         def __init__(self, 初值=0):
#             self.值 = 初值
#
#     计数器(5)         ⇒  造对象时自动调用 __init__，把 值 设成 5
#
#   ⚠ 名字是**前后各两个下划线**：`__init__`
#     写成 `_init_`（一个）或 `__int__`（少了 i）都不会报错，
#     但它就只是个普通方法，造对象时**不会被调用** ——
#     于是你会奇怪"为什么属性不存在"。
#
#   ⚠ `__init__` **不该写 return**（它负责准备，不负责交结果）。
#
# ─────────────────────────────────────────────────────
#  `self` 是什么
# ─────────────────────────────────────────────────────
#
#      def 存(self, n):
#          self.余额 += n
#          ↑
#          "我自己"
#
#   `self.余额` 是**这个对象自己的**余额。
#   如果写成 `余额`（没有 self.），那是一个**局部变量**，
#   跟对象毫无关系 —— 新手最常混这一点。
#
#   ⚠ 调用时**不用传 self**：
#        定义：def 存(self, n):
#        调用：罐.存(50)            ← Python 自己把"罐"塞进 self
#
# ─────────────────────────────────────────────────────
#  ⚠ 类名后面的括号别忘
# ─────────────────────────────────────────────────────
#
#     罐 = 存钱罐           ❌ 这只是给**类**起了个别名，没造对象
#     罐 = 存钱罐()         ✅ 这一对括号才是"造一个"
#
# ─────────────────────────────────────────────────────
#  这一课只讲这一件事
# ─────────────────────────────────────────────────────
#   封装（把规则关进方法里、不让外面乱改）是第 16 课。
#   这一课只要"能造出一个带属性的对象"就够了。
#
# ─────────────────────────────────────────────────────
#  轮到你了 —— 在答题文件里写这几件事
# ─────────────────────────────────────────────────────
#   写一个类 学生：
#     · 造出来时接收 姓名
#     · 属性 `姓名` 存住它
#     · 方法 `问好()` 返回 f"我是{姓名}"
#
#   ⚠ 就这些。不用检查余额、不用管规则 —— 那是第 16 课。
#
# ─────────────────────────────────────────────────────

# 第 16 课：封装 —— 把一个东西的规则关进它自己里面

# ─────────────────────────────────────────────────────
#  先介绍：什么叫"规则只有一个出口"
# ─────────────────────────────────────────────────────
#
# 第 15 课你学会了造对象。但光有属性还不够 —— 看这个：
#
#     class 存钱罐:
#         def __init__(self, 初值=0):
#             self.余额 = 初值
#
#     罐 = 存钱罐(100)
#     罐.余额 = -999          ← ⚠ 外面随便就能把余额改成负数！
#
# 问题：`余额` 是公开的，**谁都能乱改** —— 你在方法里写多少检查都没用。
#
# 封装就是：**把"改数据"这件事收进方法里，外面只能通过方法改。**
#
#     class 存钱罐:
#         def __init__(self, 初值=0):
#             self._余额 = 初值          ← 下划线开头 = "别从外面碰它"
#
#         @property
#         def 余额(self):                ← 外面读 罐.余额 会走到这里
#             return self._余额
#
#         def 存(self, n):
#             self._余额 += n
#
#         def 取(self, n):
#             if n > self._余额:          ← 规则就在这里，跑不掉
#                 return False
#             self._余额 -= n
#             return True
#
#   ⇒ 想把余额改成负数？做不到 —— 唯一的入口是 `取`，而 `取` 会检查。
#
#   ⚠ "封装"不要理解成"藏起来"。它的真实意思是：
#     **一个东西的规则只有一个出口。**
#
# ─────────────────────────────────────────────────────
#  ⚠⚠ 这一课唯一要记住的顺序：先检查，通过才动
# ─────────────────────────────────────────────────────
#
# 新手最容易这么写：
#
#     def 取(self, n):
#         self.余额 -= n              ← ❌ 已经扣了！
#         if self.余额 < 0:
#             return False
#
# 这时余额**已经被改坏了** —— 你返回 False 也晚了。
#
#   ⇒ 顺序永远是：**先检查，通过才动。**
#
#     def 取(self, n):
#         if n > self.余额:
#             return False            ← 什么都没动
#         self.余额 -= n              ← 检查过了才扣
#         return True
#
#   ⚠ 边界：`n > self.余额` 是"大于才拒绝"。
#     写成 `>=` 的话，"正好取光"（取 100，余额 100）会被拒绝 —— 那是错的。
#
# ─────────────────────────────────────────────────────
#  这一课只讲这一件事
# ─────────────────────────────────────────────────────
#   这一课只讲"用方法管住数据"。继承、接口这些以后再说。
#
# ─────────────────────────────────────────────────────
#  轮到你了 —— 在答题文件里写这几件事
# ─────────────────────────────────────────────────────
#   写一个类 存钱罐：
#     · 造出来时有余额（默认 0）
#     · 方法 `存(n)`：增加余额
#     · 方法 `取(n)`：够就扣掉并返回 True，**不够就返回 False 且余额一点都不变**
#     · 属性 `余额`：外面能读，但不能直接改
#
# ─────────────────────────────────────────────────────
#  怎么自己测
# ─────────────────────────────────────────────────────
#     python -m hybridengine_light 检查 16

"""第 16 课参考答案 —— 封装。"""


class 存钱罐:
    def __init__(self, 初值=0):
        # ⚠ 下划线开头 = "别从外面碰它"
        self._余额 = 初值

    @property
    def 余额(self):
        # ⚠ @property 让 `罐.余额` 读得到，但**赋不了值**
        #   ⇒ 想改余额只有 存() / 取() 两个入口，而它们会检查
        return self._余额

    def 存(self, n):
        self._余额 += n

    def 取(self, n):
        # ⚠⚠ 顺序是关键：**先检查，通过才动**。
        #   反过来的话，一次失败的取款已经把余额改坏了。
        # ⚠ 边界是 `>` 不是 `>=` —— 正好取光要能成功。
        if n > self._余额:
            return False
        self._余额 -= n
        return True
