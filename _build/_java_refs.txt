static int 求和(int n) {
    int 和 = 0;                      // ← 累加变量必须有初值
    for (int i = 1; i <= n; i++) {   // ← 是 <= n，不是 < n（n = 1 时要跑一次）
        和 += i;
    }
    return 和;
}
static String 重复(String 文字, int 次数) {
    String 结果 = "";
    for (int i = 0; i < 次数; i++) {   // ← 这一次是从 0 数起，所以用 < 次数
        结果 += 文字;
    }
    return 结果;
}
static int 数位数(int n) {
    int 位 = 0;
    int 剩 = n;
    do {                     // ← 先做一遍再看条件：n = 0 也要数出 1 位
        位++;
        剩 /= 10;
    } while (剩 > 0);
    return 位;
}
static int 最小因数(int n) {
    if (n < 2) {
        return -1;
    }
    int 结果 = n;                    // n 是质数的话，答案就是它自己
    int 试 = 2;
    while (试 < n) {
        if (n % 试 == 0) {
            结果 = 试;
            break;                   // 找到最小的就跳出，不用再找了
        }
        试++;
    }
    return 结果;
}
static int 偶数个数(int[] 数字) {
    int 个数 = 0;
    for (int x : 数字) {          // 增强 for：把元素一个一个拿出来
        if (x % 2 != 0) {
            continue;             // 奇数跳过，不计数
        }
        个数++;
    }
    return 个数;
}

        System.out.println("名字还是=" + 甲.取名字());

        人 乙 = 甲;                     // ⚠ 这不是 new —— 两个名字指同一块地方
        乙.改年龄(40);                  // 通过 乙 改
        System.out.println("第一个名字看到=" + 甲.取年龄());   // 甲 看到的也变了
    }

    private int 分数;

    public 学生(String 名字, int 分数, String 班级) {
        if (名字 == null || 名字.trim().isEmpty()) {
            this.名字 = "无名";
        } else {
            this.名字 = 名字;
        }
        if (分数 < 0) {
            分数 = 0;
        }
        if (分数 > 100) {
            分数 = 100;
        }
        this.分数 = 分数;
        if (班级 == null || 班级.trim().isEmpty()) {

            this.班级 = 班级;

        this(名字, 0, "未分班");
    }

public class Main {
    public static void main(String[] args) {
        长方形 桌 = new 长方形(4, 5);
        圆 盘 = new 圆(2);
        System.out.println(桌.说明());            // 面积=20

        形状 一个形状 = new 圆(2);
        System.out.println(一个形状.说明());       // 面积=13，跑的是 圆 的 面积()
        形状[] 图形们 = new 形状[2];              // 元素类型是父类
        图形们[0] = new 长方形(4, 5);
        图形们[1] = new 圆(2);
        for (形状 图 : 图形们) {
            System.out.println(图.说明());         // 动态绑定：各自找对自己的 面积()
        }
    }
}
abstract class 形状 {
    public abstract double 面积();
    public String 说明() {
        return "面积=" + Math.round(面积());
    }
}

        this.高 = 高;
    }
    @Override
    public double 面积() {
        return 宽 * 高;
    }
}

    public 圆(double 半径) {
        this.半径 = 半径;
    }
    @Override
    public double 面积() {

    }

static String 每行和(int[][] 格, int 行数) {
    StringBuilder 结果 = new StringBuilder("[");
    for (int i = 0; i < 行数; i++) {
        int 和 = 0;
        for (int j = 0; j < 格[i].length; j++) {   /* ⚠ 不是 格[0].length */
            和 = 和 + 格[i][j];
        }
        if (i > 0) {
            结果.append(",");
        }
        结果.append(和);
    }
    结果.append("]");
    return 结果.toString();
}
static int 列和(int[][] 格, int 行数, int 列号) {
    int 和 = 0;
    for (int i = 0; i < 行数; i++) {
        if (列号 < 格[i].length) {                  /* ⚠ 问的是第 i 行 */
            和 = 和 + 格[i][列号];
        }
    }
    return 和;
}
static int 主对角线和(int[][] 格) {
    int 最短 = 格.length;
    for (int i = 0; i < 格.length; i++) {
        if (格[i].length < 最短) {
            最短 = 格[i].length;
        }
    }
    int 和 = 0;
    for (int i = 0; i < 最短; i++) {
        和 = 和 + 格[i][i];
    }
    return 和;
}
static int 矩阵最大(int[][] 格) {

static int 因数个数(int n) {

        return 0;                          /* 0 和负数的因数有无限多个，不算 */
    }

    for (int i = 1; (long) i * i <= n; i++) {
        if (n % i == 0) {
            个数 = 个数 + 1;                /* i 是一个因数 */
            if (i != n / i) {
                个数 = 个数 + 1;            /* n/i 是另一个（完全平方数时它俩相等，只算一次） */
            }
        }
    }
    return 个数;
}

    if (n < 2) {
        return "" + n;                     /* 1、0、负数原样返回 */
    }

            结果.append(i);

            结果.append("*");
        }
        结果.append(剩);
    }
    return 结果.toString();
}

static long 最大公约数(long a, long b) {

    b = Math.abs(b);
    if (b == 0) {
        return a;                          /* 余数是 0 了 —— 基准情形，也顺手管住了 gcd(0, b) = b */
    }
    return 最大公约数(b, a % b);             /* 换成「更小的一版」再问 */
}
static long 最小公倍数(long a, long b) {
    a = Math.abs(a);
    b = Math.abs(b);
    if (a == 0 || b == 0) {
        return 0;                          /* 0 和任何数的「公倍数」这一课规定是 0 */
    }
    return (a / 最大公约数(a, b)) * b;        /* ← 先除，再乘 */
}

static int 二分查找(int[] 数, int 目标) {
    int low = 0;
    int high = 数.length - 1;                 /* ⚠ length - 1：下标从 0 开始 */
    while (low <= high) {                     /* ⚠ <= ：区间里还剩一个数时也要进来比一比 */
        int mid = low + (high - low) / 2;     /* 这样写不会溢出（low + high 在大数组上会） */
        if (数[mid] == 目标) {
            return mid;
        }
        if (数[mid] < 目标) {
            low = mid + 1;                    /* ⚠ mid + 1：把 mid 排除掉，否则死循环 */
        } else {
            high = mid - 1;                   /* ⚠ mid - 1 */
        }
    }
    return -1;                                /* 区间空了 ⇒ 就是没有 */
}

static void 归并(int[] 数, int 左, int 中, int 右) {
    int[] 临时 = new int[右 - 左 + 1];
    int i = 左;
    int j = 中 + 1;
    int k = 0;
    while (i <= 中 && j <= 右) {
        /* ⚠ 相等时取左边的（<=），归并排序才是稳定的 */
        if (数[i] <= 数[j]) {
            临时[k] = 数[i];
            i = i + 1;
        } else {
            临时[k] = 数[j];
            j = j + 1;
        }
        k = k + 1;
    }
    while (i <= 中) {                     /* 左段还剩的（它们本来就都比右边剩的大） */
        临时[k] = 数[i];
        i = i + 1;
        k = k + 1;
    }
    while (j <= 右) {                     /* 右段还剩的 */
        临时[k] = 数[j];
        j = j + 1;
        k = k + 1;
    }
    for (int t = 0; t < 临时.length; t++) {   /* 写回原数组的对应位置 */
        数[左 + t] = 临时[t];
    }
}
static void 归并排序(int[] 数, int 左, int 右) {
    if (左 >= 右) {
        return;                           /* 基准情形：0 个或 1 个数，已经有序 */
    }
    int 中 = 左 + (右 - 左) / 2;            /* 这样写不会溢出（第 26 课讲过） */
    归并排序(数, 左, 中);                    /* 治左半 */
    归并排序(数, 中 + 1, 右);                /* 治右半 */
    归并(数, 左, 中, 右);                    /* ⚠ 合必须在两个递归**之后** —— 两半得先有序 */
}
static int 分治最大(int[] 数, int 左, int 右) {
    if (左 > 右) {
        return Integer.MIN_VALUE;         /* 空区间：返回最小的 int，当"没有"用 */
    }
    if (左 == 右) {
        return 数[左];                     /* 基准情形：一个数，它就是最大的 */
    }
    int 中 = 左 + (右 - 左) / 2;
    int 左大 = 分治最大(数, 左, 中);
    int 右大 = 分治最大(数, 中 + 1, 右);
    return 左大 > 右大 ? 左大 : 右大;
}

static int 划分(int[] 数, int 左, int 右) {
    /* ① 随机挑一个当基准，先换到末尾 —— 这样下面的 Lomuto 逻辑不用动 */
    int 随 = 左 + 随机.nextInt(右 - 左 + 1);
    int 临时 = 数[随];
    数[随] = 数[右];
    数[右] = 临时;
    /* ② Lomuto：小 记录"比基准小的那一段"的末尾 */
    int 基准 = 数[右];
    int 小 = 左 - 1;
    for (int i = 左; i < 右; i++) {
        if (数[i] <= 基准) {
            小 = 小 + 1;
            int 换 = 数[小];
            数[小] = 数[i];
            数[i] = 换;
        }
    }
    /* ③ 把基准放到分界点上 —— 这一步之后它的位置就是最终的了 */
    int 换2 = 数[小 + 1];
    数[小 + 1] = 数[右];
    数[右] = 换2;
    return 小 + 1;
}

static boolean 括号匹配(String s) {
    Deque<Character> 栈 = new ArrayDeque<>();
    for (int i = 0; i < s.length(); i++) {
        char c = s.charAt(i);
        if (c == '(' || c == '[' || c == '{') {
            栈.push(c);
        } else if (c == ')' || c == ']' || c == '}') {
            if (栈.isEmpty()) {
                return false;                     /* 右括号来了，可没有左括号等它 */
            }
            char 顶 = 栈.pop();                    /* 最近的那个左括号 */
            if (顶 == '(' && c != ')') return false;
            if (顶 == '[' && c != ']') return false;
            if (顶 == '{' && c != '}') return false;
        }
    }
    return 栈.isEmpty();                          /* 还有左括号没配对上 ⇒ 错 */
}
static String 用栈倒序(int[] 数) {
    Deque<Integer> 栈 = new ArrayDeque<>();
    for (int i = 0; i < 数.length; i++) {
        栈.push(数[i]);                        /* 后进 */
    }
    StringBuilder 结果 = new StringBuilder();
    while (!栈.isEmpty()) {                      /* ⚠ 先问空不空：空栈 pop 得到 null */
        if (结果.length() > 0) {
            结果.append(" ");
        }
        结果.append(栈.pop());                  /* 先出 */
    }
    return 结果.toString();
}
static String 用队列轮转(int[] 数, int k) {

    if (终行 < 0 || 终行 >= 行数 || 终列 < 0 || 终列 >= 列数) return false;
    if (迷宫[起行][起列] == 1 || 迷宫[终行][终列] == 1) return false;

    Deque<int[]> 队 = new ArrayDeque<>();
    队.offer(new int[]{起行, 起列});
    来过[起行][起列] = true;                  /* ⚠ 一进队列就标记，别等出队 */

            return true;

static long 爬楼梯(int n) {
    if (n == 0) {
        return 1;                        /* 站在原地不动 —— 算一种"走法" */
    }
    long[] dp = new long[n + 1];
    dp[0] = 1;
    dp[1] = 1;
    for (int i = 2; i <= n; i++) {
        dp[i] = dp[i - 1] + dp[i - 2];
    }
    return dp[n];
}
static long 打家劫舍(int[] 金额) {
    int 家数 = 金额.length;
    if (家数 == 0) {
        return 0;
    }
    long[] dp = new long[家数 + 1];
    dp[0] = 0;
    dp[1] = 金额[0];                      /* 只有一家 ⇒ 就偷它 */
    for (int i = 2; i <= 家数; i++) {
        long 偷这家 = 金额[i - 1] + dp[i - 2];   /* 偷了它，i-1 家就不能偷 */
        long 不偷 = dp[i - 1];
        dp[i] = 偷这家 > 不偷 ? 偷这家 : 不偷;
    }
    return dp[家数];
}
static int 背包(int[] 重量, int[] 价值, int 容量) {
    int[] dp = new int[容量 + 1];         /* 初始全 0：容量 0 装不下任何东西 */
    for (int i = 0; i < 重量.length; i++) {
        for (int j = 容量; j >= 重量[i]; j--) {
            int 拿这件 = 价值[i] + dp[j - 重量[i]];
            if (拿这件 > dp[j]) {
                dp[j] = 拿这件;
            }
        }
    }
    return dp[容量];
}

static long 网格路径数(int 行数, int 列数) {
    if (行数 <= 0 || 列数 <= 0) {
        return 0;
    }
    long[][] dp = new long[行数][列数];
    for (int i = 0; i < 行数; i++) {
        dp[i][0] = 1;                     /* ⚠ 第一列：只能一路往下走 ⇒ 1 种 */
    }
    for (int j = 0; j < 列数; j++) {
        dp[0][j] = 1;                     /* ⚠ 第一行：只能一路往右走 ⇒ 1 种 */
    }
    for (int i = 1; i < 行数; i++) {
        for (int j = 1; j < 列数; j++) {
            dp[i][j] = dp[i - 1][j] + dp[i][j - 1];   /* 上面 + 左边 */
        }
    }
    return dp[行数 - 1][列数 - 1];
}
static int 最长公共子序列(String 甲, String 乙) {
    int 甲长 = 甲.length();
    int 乙长 = 乙.length();
    /* ⚠ 表要比字符串长一格：dp[i][j] 说的是"前 i 个"，i 要到 length() */
    int[][] dp = new int[甲长 + 1][乙长 + 1];
    for (int i = 1; i <= 甲长; i++) {
        for (int j = 1; j <= 乙长; j++) {
            if (甲.charAt(i - 1) == 乙.charAt(j - 1)) {   /* ⚠ 第 i 个字符的下标是 i-1 */
                dp[i][j] = dp[i - 1][j - 1] + 1;
            } else {
                int 扔甲 = dp[i - 1][j];
                int 扔乙 = dp[i][j - 1];
                dp[i][j] = 扔甲 > 扔乙 ? 扔甲 : 扔乙;      /* ⚠ 这一支不能少 */
            }
        }
    }
    return dp[甲长][乙长];
}

static long 最短距离(List<List<int[]>> 图, int 起, int 终) {
    if (图 == null || 图.isEmpty()) return -1;
    int 点数 = 图.size();
    if (起 < 0 || 起 >= 点数 || 终 < 0 || 终 >= 点数) return -1;
    if (起 == 终) return 0;
    long 无穷 = Long.MAX_VALUE / 4;         /* 别用 MAX_VALUE：加权重时会溢出 */
    long[] 最短 = new long[点数];
    Arrays.fill(最短, 无穷);
    最短[起] = 0;
    /* ⚠ 比较器写 Long.compare，不要写 a[0] - b[0]（long 相减会溢出） */
    PriorityQueue<long[]> 堆 = new PriorityQueue<>((a, b) -> Long.compare(a[0], b[0]));
    堆.offer(new long[]{0, 起});
    while (!堆.isEmpty()) {
        long[] 当前 = 堆.poll();
        long 距离 = 当前[0];
        int 点 = (int) 当前[1];
        if (距离 > 最短[点]) continue;       /* ⚠ 堆里的过期条目，跳过 */
        if (点 == 终) return 距离;            /* 第一个被取出来的终点就是最短的 */
        for (int[] 一条 : 图.get(点)) {
            int 邻居 = 一条[0];
            long 新距离 = 距离 + 一条[1];
            if (新距离 < 最短[邻居]) {        /* ⚠ 只有更近才更新 —— 这就是"松弛" */
                最短[邻居] = 新距离;
                堆.offer(new long[]{新距离, 邻居});
            }
        }
    }
    return 最短[终] == 无穷 ? -1 : 最短[终];
}
static String 路径(List<List<int[]>> 图, int 起, int 终) {
    if (图 == null || 图.isEmpty()) return "空";
    int 点数 = 图.size();
    if (起 < 0 || 起 >= 点数 || 终 < 0 || 终 >= 点数) return "空";
    if (起 == 终) return "" + 起;
    long 无穷 = Long.MAX_VALUE / 4;
    long[] 最短 = new long[点数];
    int[] 前驱 = new int[点数];
    Arrays.fill(最短, 无穷);
    Arrays.fill(前驱, -1);
    最短[起] = 0;
    PriorityQueue<long[]> 堆 = new PriorityQueue<>((a, b) -> Long.compare(a[0], b[0]));
    堆.offer(new long[]{0, 起});
    while (!堆.isEmpty()) {
        long[] 当前 = 堆.poll();
        long 距离 = 当前[0];
        int 点 = (int) 当前[1];
        if (距离 > 最短[点]) continue;
        for (int[] 一条 : 图.get(点)) {
            int 邻居 = 一条[0];
            long 新距离 = 距离 + 一条[1];
            if (新距离 < 最短[邻居]) {
                最短[邻居] = 新距离;
                前驱[邻居] = 点;              /* ⚠ 记下是从谁过来的 */
                堆.offer(new long[]{新距离, 邻居});
            }
        }
    }
    if (最短[终] == 无穷) return "空";
    /* 从终点一路往起点回溯，再翻过来 */
    List<String> 倒着 = new ArrayList<>();
    int 走 = 终;
    while (走 != -1) {
        倒着.add("" + 走);
        if (走 == 起) break;                 /* ⚠ 到起点就停（起点也要放进路径里） */
        走 = 前驱[走];
    }
    StringBuilder 串 = new StringBuilder();
    for (int i = 倒着.size() - 1; i >= 0; i--) {
        if (串.length() > 0) 串.append(" ");
        串.append(倒着.get(i));
    }
    return 串.toString();
}
