答案|// ② 平均分
答案|double 平均分(const std::vector<成绩>& 们) {
答案|    if (们.empty()) return 0.0;          // 空的时候先返回，别等到除零
答案|    int 总和 = 0;
答案|    for (const 成绩& 一 : 们) {           // ⚠ 加 & 就不复制；加 const 表示只读
答案|        总和 += 一.分数;
答案|    }
答案|    // ⚠⚠ 至少一边转成 double，否则这里是**整数除法**（小数被丢掉）
答案|    return (double)总和 / (int)们.size();
答案|}

答案|std::string 最高分名字(const std::vector<成绩>& 们) {

答案|    }
